cmake_minimum_required(VERSION 3.22.1)
project(fbr_speech)
if(NOT FBR_WORKSPACE)
  message(FATAL_ERROR "FBR_WORKSPACE must identify the prepared source worktree")
endif()
file(TO_CMAKE_PATH "${FBR_WORKSPACE}" FBR_WORKSPACE)
set(FBR_SUPPORT "${FBR_WORKSPACE}/build-support/android-speech")
set(PIPER_ANDROID_UPSTREAM "${FBR_SUPPORT}/piper-kotlin/piper-engine/src/main/cpp")
set(PIPER1_GPL_DIR "${FBR_SUPPORT}/piper-adapted")
set(ESPEAK_NG_DIR "${FBR_SUPPORT}/piper-kotlin/third-party/espeak-ng")
set(SONIC_DIR "${FBR_SUPPORT}/piper-kotlin/third-party/sonic")
if(NOT ONNXRUNTIME_DIR)
  set(ONNXRUNTIME_DIR "${FBR_SUPPORT}/onnxruntime-1.22.0")
endif()
file(TO_CMAKE_PATH "${ONNXRUNTIME_DIR}" ONNXRUNTIME_DIR)
if(WIN32)
  set(PATCH_CMD "C:/Program Files/Git/usr/bin/patch.exe" CACHE FILEPATH "Verified Git patch utility")
endif()
add_subdirectory("${PIPER_ANDROID_UPSTREAM}" "${CMAKE_CURRENT_BINARY_DIR}/piper-native")
target_sources(piper_jni PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/speech_bridge.cpp")
target_include_directories(piper_jni PRIVATE "${PIPER_ANDROID_UPSTREAM}")
# Android 15+ can use 16 KiB pages. Inspect the prebuilt ORT separately as well.
target_link_options(piper_jni PRIVATE "-Wl,-z,max-page-size=16384")
