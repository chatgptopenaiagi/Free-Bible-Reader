cmake_minimum_required(VERSION 3.22.1)
project(fbr_ro_voice LANGUAGES C CXX)
if(NOT FBR_WORKSPACE)
  message(FATAL_ERROR "FBR_WORKSPACE must identify the prepared worktree")
endif()
file(TO_CMAKE_PATH "${FBR_WORKSPACE}" FBR_WORKSPACE)
set(ESPEAK "${FBR_WORKSPACE}/build-support/android-speech/piper-kotlin/third-party/espeak-ng")
if(NOT EXISTS "${ESPEAK}/src/libespeak-ng/espeak_api.c")
  message(FATAL_ERROR "Prepare the existing pinned eSpeak checkout first")
endif()
set(ESPEAK_NAMES common mnemonics error ieee80 compiledata compiledict dictionary encoding intonation langopts numbers phoneme phonemelist readclause setlengths soundicon spect ssml synthdata synthesize tr_languages translate translateword voices wavegen speech espeak_api klatt)
set(ESPEAK_SOURCES "")
foreach(name IN LISTS ESPEAK_NAMES)
  list(APPEND ESPEAK_SOURCES "${ESPEAK}/src/libespeak-ng/${name}.c")
endforeach()
file(GLOB UCD_SOURCES "${ESPEAK}/src/ucd-tools/src/*.c")
add_library(fbr_ro_voice SHARED ro_voice.cpp ${ESPEAK_SOURCES} ${UCD_SOURCES})
target_compile_features(fbr_ro_voice PRIVATE cxx_std_17)
target_include_directories(fbr_ro_voice PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ESPEAK}/src/include" "${ESPEAK}/src/libespeak-ng" "${ESPEAK}/src/ucd-tools/src/include")
target_compile_definitions(fbr_ro_voice PRIVATE HAVE_CONFIG_H ESPEAK_NG_API= LIBESPEAK_NG_EXPORT=1 N_PATH_HOME=1024 "PATH_ESPEAK_DATA=\"/not-used/espeak-ng-data\"")
target_link_libraries(fbr_ro_voice PRIVATE android log)
target_link_options(fbr_ro_voice PRIVATE "-Wl,-z,max-page-size=16384")
